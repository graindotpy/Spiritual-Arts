# Foundry bridge integration environment variables

# Existing Discord destination (unchanged)
# DISCORD_WEBHOOK_URL=https://discord.com/api/webhooks/...

# Foundry destination toggle (optional; defaults to enabled when URL is set)
# FOUNDRY_ENABLED=true

# Example: http://127.0.0.1:30555/webhook
# FOUNDRY_WEBHOOK_URL=

# Shared Bearer token sent in Authorization header
# Must match the bridge receiver and Foundry module shared secret.
# FOUNDRY_WEBHOOK_TOKEN=
